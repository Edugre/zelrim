cmake_minimum_required(VERSION 3.20)
project(zelrim VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "Zelrim transport requires Windows")
endif()
add_library(zelrim_bridge bridge/bridge.cpp)
target_include_directories(zelrim_bridge PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_features(zelrim_bridge PUBLIC cxx_std_20)
target_compile_definitions(zelrim_bridge PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX)
if(MSVC)
    target_compile_options(zelrim_bridge PRIVATE /W4 /permissive-)
endif()
add_executable(skyrim_bridge_test tools/endpoint.cpp)
target_compile_definitions(skyrim_bridge_test PRIVATE ZELRIM_SKYRIM_SIDE)
target_link_libraries(skyrim_bridge_test PRIVATE zelrim_bridge)
add_executable(oot_bridge_test tools/endpoint.cpp)
target_link_libraries(oot_bridge_test PRIVATE zelrim_bridge)
add_executable(bridge_monitor tools/monitor.cpp)
target_link_libraries(bridge_monitor PRIVATE zelrim_bridge)
option(ZELRIM_BUILD_SKSE "Build the read-only Zelrim SKSE plugin" OFF)
if(ZELRIM_BUILD_SKSE)
    add_subdirectory(integrations/skse)
endif()
include(CTest)
if(BUILD_TESTING)
    add_subdirectory(tests/shipwright-host)
    add_executable(bridge_tests tests/bridge_tests.cpp)
    target_sources(bridge_tests PRIVATE integrations/skse/world_telemetry.cpp)
    target_link_libraries(bridge_tests PRIVATE zelrim_bridge)
    add_test(NAME bridge_lifecycle COMMAND bridge_tests "$<TARGET_FILE:skyrim_bridge_test>" "$<TARGET_FILE:oot_bridge_test>")
    set_tests_properties(bridge_lifecycle PROPERTIES TIMEOUT 45 RUN_SERIAL TRUE)
    add_test(NAME shipwright_adapter_lifecycle COMMAND bridge_tests "$<TARGET_FILE:skyrim_bridge_test>" "$<TARGET_FILE:soh>")
    set_tests_properties(shipwright_adapter_lifecycle PROPERTIES TIMEOUT 45 RUN_SERIAL TRUE)
    if(ZELRIM_BUILD_SKSE)
        add_executable(skse_adapter_host tests/skse_host.cpp)
        target_include_directories(skse_adapter_host PRIVATE "${ZELRIM_SKSE_SOURCE}")
        target_link_libraries(skse_adapter_host PRIVATE zelrim_bridge)
        target_compile_definitions(skse_adapter_host PRIVATE "ZELRIM_TEST_DLL=L\"$<TARGET_FILE:Zelrim>\"")
        add_dependencies(skse_adapter_host Zelrim)
        add_test(NAME skse_adapter_tasks COMMAND skse_adapter_host --self-test)
        add_test(NAME both_adapters_lifecycle COMMAND bridge_tests "$<TARGET_FILE:skse_adapter_host>" "$<TARGET_FILE:soh>")
        set_tests_properties(skse_adapter_tasks both_adapters_lifecycle PROPERTIES TIMEOUT 45 RUN_SERIAL TRUE)
    endif()
endif()
